### 9.3 可复跑的对照程序

把下面脚本存为文件后在仓库根运行 `python3 <file> README.md`(或
`docs/08-architecture.md`、`docs/C-acceptance.md`)。它从 `lib.rs` 的 clap 声明重建
"顶层 → 叶子"映射,再扫描文档反引号里的命令 token,输出 A 类(代码有、文档未提)与
B 类(文档写成不存在的叶子)。斜杠连写(`a/b/c`)、`{a|b}` 简写会被当作未知叶子,
属于该启发式的已知误报。

```python
import re, sys, hashlib
LIB = "crates/vantage-cli/src/lib.rs"
src = open(LIB, encoding="utf-8").read().splitlines()

enums, cur = {}, None
for line in src:
    m = re.match(r"^pub enum (\w+) \{", line)
    if m:
        cur = m.group(1); enums[cur] = []; continue
    if cur:
        if line.startswith("}"):
            cur = None; continue
        m = re.match(r"^    ([A-Za-z_]\w*)\s*([,{]?)", line)
        if m and not line.strip().startswith("//"):
            enums[cur].append(m.group(1))

kebab = lambda n: re.sub(r"(?<!^)(?=[A-Z])", "-", n).lower()
top = {kebab(n): [] for n in enums.get("Command", [])}
sub = {
    "battery": "BatteryCommand", "usb": "UsbCommand", "power": "PowerCommand",
    "perf": "PerfCommand", "tune": "TuneCommand", "kbd": "KbdCommand",
    "touchpad": "TouchpadAction", "panel": "PanelCommand",
    "privacy": "PrivacyCommand", "sense": "SenseCommand", "audio": "AudioCommand",
    "bios": "BiosCommand", "magicbay": "MagicbayCommand", "osd": "OsdCommand",
    "hotkey": "HotkeyCommand", "update": "UpdateCommand", "scan": "ScanCommand",
    "snapshot": "SnapshotCommand", "wifi": "WifiCommand", "radio": "RadioCommand",
    "fix": "FixCommand", "scene": "SceneCommand", "daemon": "DaemonCommand",
    "cpu": "CpuCommand", "completions": "Shell",
}
for name, en in sub.items():
    if name in top and en in enums:
        top[name] = [kebab(n) for n in enums[en]]
print("lib.rs", hashlib.sha256(open(LIB, "rb").read()).hexdigest()[:8],
      len(src), "lines", len(top), "top-level")

text = open(sys.argv[1], encoding="utf-8").read()
found, unknown = set(), set()
for span in re.findall(r"`([^`\n]+)`", text):
    toks = span.strip().split()
    if toks and toks[0] == "vantage":
        toks = toks[1:]
    if not toks:
        continue
    head = toks[0]
    if head not in top:
        continue
    found.add(head)
    rest = [t for t in toks[1:] if not t.startswith("-") and not t.startswith("<")]
    if rest and "/" not in head:
        leaf = rest[0].rstrip(",").split("|")[0]
        if top[head] and leaf not in top[head]:
            unknown.add((head, leaf, span[:48]))

print("\nA 类(代码有、文档未提):")
for k in sorted(top):
    if top[k] and k not in found:
        print("  ", k, "->", " ".join(top[k]))
print("\nB 类(文档写成不存在的叶子,括号内为例句):")
for item in sorted(unknown):
    print("  ", item)
```
